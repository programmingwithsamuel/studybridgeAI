# Course codes are normalized to handle spaces and hyphens.
# C0S101 in the PDF is treated as COS 101 because the
# existing curriculum record uses the letter O.


def normalize_code(code: str) -> str:
    return "".join(character for character in code.upper() if character.isalnum())

def main():
    db = SessionLocal()

    try:
        courses = db.query(Course).all()

        course_lookup = {
            normalize_code(course.code): course
            for course in courses
        }

        # Some curriculum entries include the NAU prefix.
        aliases = {
            "COS101": ["COS101"],
            "MTH101": ["MTH101"],
            "PHY101": ["PHY101"],
            "PHY107": ["PHY107"],
            "STA111": ["STA111"],
            "NAUCSC111": ["NAUCSC111"],
            "NAUCHM101": ["NAUCHM101"],
            "NAUBIO101": ["NAUBIO101"],
            "GST111": ["GST111", "NAUGST111"],
            "GST113": ["GST113", "NAUGST113"],
        }

        resolved_courses = {}
        missing_courses = []

        for key, possible_codes in aliases.items():
            found = None

            for possible_code in possible_codes:
                found = course_lookup.get(normalize_code(possible_code))
                if found:
                    break

            if found:
                resolved_courses[key] = found
            else:
                missing_courses.append(key)

        if missing_courses:
            print("IMPORT STOPPED. These course codes were not found:")
            for code in missing_courses:
                print(f" - {code}")
            print("No topics were inserted.")
            return

        created = 0
        skipped = 0

        for course_code, topics in COURSE_TOPICS.items():
            course = resolved_courses[course_code]

            existing_topics = {
                (topic.position, topic.title)
                for topic in db.query(CourseTopic)
                .filter(CourseTopic.course_id == course.id)
                .all()
            }

            for position, (title, description) in enumerate(topics, start=1):
                if (position, title) in existing_topics:
                    skipped += 1
                    continue

                topic = CourseTopic(
                    title=title,
                    description=description,
                    position=position,
                    verification_status="unverified",
                    source=SOURCE,
                    course_id=course.id,
                )

                db.add(topic)
                created += 1

        db.commit()

        print("COURSE TOPIC IMPORT COMPLETED")
        print(f"Topics created: {created}")
        print(f"Existing topics skipped: {skipped}")
        print(f"Courses processed: {len(resolved_courses)}")

        print("\nTOPIC COUNTS:")
        for course_code, topics in COURSE_TOPICS.items():
            course = resolved_courses[course_code]
            count = (
                db.query(CourseTopic)
                .filter(CourseTopic.course_id == course.id)
                .count()
            )
            print(f"{course.code}: {count} topics")

    except Exception:
        db.rollback()
        raise

    finally:
        db.close()


if __name__ == "__main__":
    main()
